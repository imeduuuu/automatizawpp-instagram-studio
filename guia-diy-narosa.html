4.1
Crear el archivo docker-compose.yml con Evolution API + Redis
Este archivo define los servicios que se ejecutarán en el servidor
En el servidor SSH, crear el archivo (copia y pega todo)
nano /opt/narosa/docker-compose.yml
# Se abre un editor. Pega EXACTAMENTE esto (cambia los valores marcados):

version: '3.8'
services:

  evolution-api:
    image: atendai/evolution-api:latest
    container_name: evolution_api
    restart: always
    ports:
      - "8080:8080"
    environment:
      - SERVER_URL=https://bot.narosasp.com
      - AUTHENTICATION_API_KEY=CAMBIA_ESTO_POR_UNA_CLAVE_SECRETA
      - AUTHENTICATION_EXPOSE_IN_FETCH_INSTANCES=true
    volumes:
      - evolution_data:/evolution/instances
      - evolution_store:/evolution/store

  redis:
    image: redis:7-alpine
    container_name: redis
    restart: always

  n8n:
    image: n8nio/n8n:latest
    container_name: n8n
    restart: always
    ports:
      - "5678:5678"
    environment:
      - N8N_HOST=bot.narosasp.com
      - N8N_PATH=/n8n/
      - N8N_PROTOCOL=https
      - WEBHOOK_URL=https://bot.narosasp.com/n8n/
      - N8N_BASIC_AUTH_ACTIVE=true
      - N8N_BASIC_AUTH_USER=narosa
      - N8N_BASIC_AUTH_PASSWORD=CAMBIA_ESTO_POR_PASSWORD_SEGURA
    volumes:
      - n8n_data:/home/node/.n8n

volumes:
  evolution_data:
  evolution_store:
  n8n_data:

# Para guardar en nano: Ctrl+O → Enter → Ctrl+X
Arrancar todos los servicios
cd /opt/narosa && docker compose up -d
# Espera 2-3 minutos. Luego verifica que están corriendo:
docker compose ps
# Deberías ver: evolution_api, redis, n8n — todos en estado "running"
4.2
Configurar Nginx para que todo funcione con HTTPS
Nginx es el "portero" que dirige el tráfico a cada servicio
Crear archivo de configuración Nginx
nano /etc/nginx/sites-available/narosa-bot
# Pegar esto:

server {
    server_name bot.narosasp.com;

    # Evolution API (WhatsApp)
    location / {
        proxy_pass http://localhost:8080;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

    # n8n (automatización)
    location /n8n/ {
        proxy_pass http://localhost:5678/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
    }
}

# Guardar con Ctrl+O, Enter, Ctrl+X. Luego activar:
ln -s /etc/nginx/sites-available/narosa-bot /etc/nginx/sites-enabled/
nginx -t && systemctl reload nginx
# Si dice "syntax is ok" y "test is successful" → ✓ todo bien

Nodo 2 — IF Filter (evitar que el bot se responda a sí mismo)
Tipo de nodo: IF
Condición: {{ $json.body.data.key.fromMe }} is false
Esto filtra los mensajes que el bot envía → solo pasan los mensajes DE clientes
